create database corporate_db;

use corporate_db;

create table users(user_id int primary key auto_increment,name varchar(256),email varchar(256),password varchar(64),tel varchar(13),created_at datetime);


-- ordersテーブルの作成

create table orders(order_id int primary key auto_increment,user_id int not null,address varchar(256),total int, created_at datetime);

-- order_productsテーブルの作成

create table order_products(order_id int not null,item_id int,item_name varchar(64),num int,price int);
